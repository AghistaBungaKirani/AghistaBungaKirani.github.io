₊˚⊹♡ ConnectionFactory dengan pooling koneksi database ♡⊹˚₊

ConnectionFactory adalah implementasi pola Factory Method yang digunakan untuk membuat dan mengelola koneksi database. Pada contoh ini diterapkan connection pooling, yaitu teknik menyimpan koneksi yang sudah digunakan ke dalam sebuah pool agar dapat digunakan kembali tanpa harus membuat koneksi baru setiap kali diperlukan. 

//Nama: Aghista Bunga Kirani
//Nim: 251410027
//Kelas: SI2A

class DatabaseConnection {
  final int id;

  DatabaseConnection(this.id);

  void connect() {
    print("Koneksi $id digunakan");
  }
}

class ConnectionFactory {
  static final List _pool = [];
  static int _counter = 1;

  static DatabaseConnection getConnection() {
    // Jika ada koneksi di pool, gunakan kembali
    if (_pool.isNotEmpty) {
      print("Mengambil koneksi dari pool...");
      return _pool.removeLast();
    }

    // Jika tidak ada, buat koneksi baru
    print("Membuat koneksi baru...");
    return DatabaseConnection(_counter++);
  }

  static void releaseConnection(DatabaseConnection connection) {
    print("Mengembalikan koneksi ${connection.id} ke pool...");
    _pool.add(connection);
  }
}

void main() {
  var conn1 = ConnectionFactory.getConnection();
  conn1.connect();

  ConnectionFactory.releaseConnection(conn1);

  var conn2 = ConnectionFactory.getConnection();
  conn2.connect();

  print(
      "Apakah ID koneksi sama? ${conn1.id == conn2.id}");
}

₊˚⊹♡ NotificationFactory yang memilih subclass berdasarkan platform ♡⊹˚₊

NotificationFactory adalah implementasi Factory Method Pattern yang digunakan untuk membuat objek notifikasi berdasarkan platform yang dipilih. Factory akan menentukan subclass yang sesuai, seperti AndroidNotification, IOSNotification, atau WebNotification, tanpa pengguna perlu membuat objek secara langsung. Pada program ini, pengguna hanya memberikan nama platform sebagai input. Selanjutnya, NotificationFactory akan memilih dan membuat objek notifikasi yang tepat. 

//Nama: Aghista Bunga Kirani
//Nim: 251410027
//Kelas: SI2A

// Abstract Class
abstract class Notification {
  void send(String message);
}

// Subclass Android
class AndroidNotification implements Notification {
  @override
  void send(String message) {
    print("Android Notification: $message");
  }
}

// Subclass iOS
class IOSNotification implements Notification {
  @override
  void send(String message) {
    print("iOS Notification: $message");
  }
}

// Subclass Web
class WebNotification implements Notification {
  @override
  void send(String message) {
    print("Web Notification: $message");
  }
}

// Factory
class NotificationFactory {
  static Notification createNotification(String platform) {
    switch (platform.toLowerCase()) {
      case 'android':
        return AndroidNotification();

      case 'ios':
        return IOSNotification();

      case 'web':
        return WebNotification();

      default:
        throw Exception("Platform tidak didukung");
    }
  }
}

void main() {
  Notification android =
      NotificationFactory.createNotification("android");

  Notification ios =
      NotificationFactory.createNotification("ios");

  Notification web =
      NotificationFactory.createNotification("web");

  android.send("Pesan untuk Android");
  ios.send("Pesan untuk iOS");
  web.send("Pesan untuk Web");
}

₊˚⊹♡ ShapeFactory dengan cache untuk shapes yang sama ♡⊹˚₊

ShapeFactory merupakan implementasi Factory Method Pattern yang digunakan untuk membuat objek bentuk (shape) seperti Circle dan Square. Factory akan memilih subclass yang sesuai berdasarkan jenis shape yang diminta oleh pengguna. Pada program ini diterapkan cache (Flyweight) untuk menyimpan objek shape yang sudah pernah dibuat. Jika pengguna meminta shape dengan jenis dan ukuran yang sama, factory tidak akan membuat objek baru, melainkan mengembalikan objek yang sudah tersimpan di dalam cache. Hal ini dapat menghemat penggunaan memori dan meningkatkan efisiensi program. 

//Nama: Aghista Bunga Kirani
//Nim: 251410027
//Kelas: SI2A

// Abstract Class
abstract class Shape {
  void draw();
}

// Subclass Circle
class Circle implements Shape {
  final double radius;

  Circle(this.radius);

  @override
  void draw() {
    print("Menggambar Circle dengan radius $radius");
  }
}

// Subclass Square
class Square implements Shape {
  final double side;

  Square(this.side);

  @override
  void draw() {
    print("Menggambar Square dengan sisi $side");
  }
}

// Factory dengan Cache
class ShapeFactory {
  static final Map _cache = {};

  static Shape createShape(String type, double size) {
    // Validasi
    if (size <= 0) {
      throw Exception("Ukuran harus lebih dari 0");
    }

    String key = "$type-$size";

    // Cek cache
    if (_cache.containsKey(key)) {
      print("Mengambil shape dari cache...");
      return _cache[key]!;
    }

    Shape shape;

    switch (type.toLowerCase()) {
      case "circle":
        shape = Circle(size);
        break;

      case "square":
        shape = Square(size);
        break;

      default:
        throw Exception("Shape tidak tersedia");
    }

    _cache[key] = shape;
    print("Membuat shape baru...");
    return shape;
  }
}

void main() {
  Shape shape1 = ShapeFactory.createShape("circle", 10);
  Shape shape2 = ShapeFactory.createShape("circle", 10);
  Shape shape3 = ShapeFactory.createShape("square", 5);

  shape1.draw();
  shape3.draw();

  print(
      "shape1 dan shape2 objek sama? ${identical(shape1, shape2)}");
}

₊˚⊹♡ Challenge: AnimalFactory ♡⊹˚₊

AnimalFactory adalah implementasi Factory Method Pattern yang digunakan untuk membuat berbagai jenis hewan berdasarkan input pengguna. Factory akan memilih subclass yang sesuai, seperti Cat atau Dog, sehingga pembuatan objek menjadi lebih terpusat dan mudah dikelola. 

//Nama: Aghista Bunga Kirani
//Nim: 251410027
//Kelas: SI2A

// Abstract Class
abstract class Animal {
  String get name;
  int get age;

  void makeSound();
}

// Subclass Cat
class Cat implements Animal {
  @override
  final String name;

  @override
  final int age;

  Cat(this.name, this.age);

  @override
  void makeSound() {
    print("$name si Kucing berkata: Meow!");
  }
}

// Subclass Dog
class Dog implements Animal {
  @override
  final String name;

  @override
  final int age;

  Dog(this.name, this.age);

  @override
  void makeSound() {
    print("$name si Anjing berkata: Woof!");
  }
}

// Factory dengan Cache
class AnimalFactory {
  static final Map _cache = {};

  static Animal createAnimal(
      String type,
      String name,
      int age,
      ) {

    // Validasi parameter
    if (name.isEmpty) {
      throw Exception("Nama tidak boleh kosong");
    }

    if (age < 0) {
      throw Exception("Umur tidak boleh negatif");
    }

    String key = "$type-$name-$age";

    // Cek cache
    if (_cache.containsKey(key)) {
      print("Mengambil hewan dari cache...");
      return _cache[key]!;
    }

    Animal animal;

    // Pemilihan subclass berdasarkan input
    switch (type.toLowerCase()) {
      case "cat":
        animal = Cat(name, age);
        break;

      case "dog":
        animal = Dog(name, age);
        break;

      default:
        throw Exception("Jenis hewan tidak tersedia");
    }

    _cache[key] = animal;
    print("Membuat hewan baru...");
    return animal;
  }
}

void main() {
  Animal animal1 =
      AnimalFactory.createAnimal("cat", "Luna", 2);

  Animal animal2 =
      AnimalFactory.createAnimal("cat", "Luna", 2);

  Animal animal3 =
      AnimalFactory.createAnimal("dog", "Max", 3);

  animal1.makeSound();
  animal3.makeSound();

  print(
      "animal1 dan animal2 objek sama? ${identical(animal1, animal2)}");
}

₊˚⊹♡ Ringkasan Materi Factory Constructor ♡⊹˚₊

Factory Constructor adalah constructor khusus dalam Dart yang digunakan untuk mengontrol proses pembuatan objek. Berbeda dengan constructor biasa yang selalu membuat objek baru, factory constructor dapat mengembalikan objek yang sudah ada, melakukan validasi, atau menentukan subclass yang akan dibuat.
